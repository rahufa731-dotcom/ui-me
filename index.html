import React, { useState } from 'react';
import { PasskeyAuth } from '@lumina/auth-sdk';

export const QuickPasskeyStep = () => {
  const [verified, setVerified] = useState(false);

  const handleAuth = async () => {
    const res = await PasskeyAuth.verifyBiometric();
    if (res.success) setVerified(true);
  };

  return (
    <button onClick={handleAuth} className="rounded-[28px] bg-gradient-to-r from-violet-600 to-pink-500">
      {verified ? "Verified ✨" : "Authorize with Passkey"}
    </button>
  );
};
